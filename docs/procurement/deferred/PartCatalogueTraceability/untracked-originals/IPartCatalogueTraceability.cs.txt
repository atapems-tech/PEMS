using PEMS.Application.DTOs;

namespace PEMS.Application.Interfaces;

/// <summary>
/// جستجوی قطعات موجود پیش از درخواست جدید و تشخیص کاندیدهای تکراری بر پایه‌ی داده‌ی واقعی دامنه
/// (درخواست‌های تکمیل‌شده‌ی کاتالوگ + Material Master). فقط‌خواندنی است و هیچ ادغام/نوشتنی انجام نمی‌دهد؛
/// در حالت ابهام تصمیم با انسان است. فقط قطعات «تأییدشده» (Completed) جزئیات می‌دهند؛ درخواست‌های در جریان
/// (که ممکن است متعلق به کاربر دیگری باشند) فقط به‌صورت شمارش اعلام می‌شوند.
/// </summary>
public interface IPartCatalogueDuplicateService
{
    Task<List<PcExistingPartHit>> SearchExistingPartsAsync(string? q, int take = 20);
    Task<PcDuplicateResult> FindDuplicateCandidatesAsync(PcDuplicateProbe probe);
}

/// <summary>پیوند قابل‌ردیابی درخواست تأییدشده به Material Master و ثبت منشأ تقاضا (FK تایپ‌دار).</summary>
public interface IPartCatalogueTraceabilityService
{
    /// <summary>MaterialItem با MaterialCode دقیقاً برابر AssignedPartNumber (یکتا) یا null؛ هیچ تغییری نمی‌دهد.</summary>
    Task<PcLookupItem?> FindLinkCandidateAsync(int requestId, PcActor actor);

    Task<PcResult> LinkMaterialItemAsync(int requestId, int materialItemId, PcActor actor);

    /// <summary>input=null ⇒ پاک‌کردن منشأ. فقط درخواست‌دهنده در Draft/Returned.</summary>
    Task<PcResult> SetDemandOriginAsync(int requestId, PcDemandOriginInput? input, PcActor actor);
}
