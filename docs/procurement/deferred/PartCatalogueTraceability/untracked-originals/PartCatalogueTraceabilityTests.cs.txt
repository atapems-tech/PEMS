/// <summary>
/// Procurement Lifecycle WP-C (Phase 1): جستجوی قطعه‌ی موجود، کاندیدهای تکراری MPN (بدون ادغام خودکار)، پیوند صریح

    private const int Owner = 1, OtherEmployee = 2;

    private sealed class Actions : IRoleModuleActionService
    {
        // نقش → مجموعه‌ی "Module.Action"
        public Dictionary<string, HashSet<string>> Grants { get; } = new();
        public Task<bool> HasActionAsync(IEnumerable<string> userRoles, string module, string action) =>
            Task.FromResult(userRoles.Any(r => Grants.TryGetValue(r, out var g) && g.Contains($"{module}.{action}")));
        public Task SeedDefaultsAsync() => Task.CompletedTask;
    }

    private sealed class Env
    {
        public required PEMSDbContext Db { get; init; }
        public required PartCatalogueDuplicateService Dup { get; init; }
        public required PartCatalogueTraceabilityService Trace { get; init; }
        // نقش‌ها: REQ = درخواست‌دهنده با Create + View سه ماژول مبدأ؛ REQNOVIEW = فقط Create؛ CAT = Review (کاتالوگ‌نویس)
        public PcActor Requester(string role = "REQ") => new(Owner, 101, new[] { role });
        public PcActor Cataloguer => new(OtherEmployee, 102, new[] { "CAT" });
        public PcActor Nobody => new(OtherEmployee, 103, new[] { "NONE" });
    }

    private static Env NewEnv()
    {
        var db = PemsDbContextFactory.CreateInMemory("pct_" + Guid.NewGuid().ToString("N"));
        var actions = new Actions();
        actions.Grants["REQ"] = new() { "PartCatalogue.Create", "WorkOrder.View", "StoreRequisition.View", "MaterialRequisition.View" };
        actions.Grants["REQNOVIEW"] = new() { "PartCatalogue.Create" };
        actions.Grants["CAT"] = new() { "PartCatalogue.Review" };
        return new Env { Db = db, Dup = new PartCatalogueDuplicateService(db), Trace = new PartCatalogueTraceabilityService(db, actions) };
    }

    private static async Task<PartCatalogueRequest> AddRequestAsync(Env e, string status, string? partNo = null, string? mpn = null,
        string? supplierPn = null, string? mfrName = null, string? oldMpn = null, string? description = null, int requester = Owner)
    {
        var r = new PartCatalogueRequest
        {
            RequesterEmployeeId = requester, Status = status, AssignedPartNumber = partNo, ManufacturerPartNumber = mpn,
            SupplierPartNumber = supplierPn, NewManufacturerName = mfrName, IsNewManufacturer = mfrName != null,
            OldManufacturerPartNumber = oldMpn, CompletedDate = status == "Completed" ? DateTime.Now : null
        };
        if (description != null)
            r.Attributes.Add(new PartCatalogueRequestAttribute { Name = PartCatalogueRequestAttribute.FreeFormatText, Value = description });
        e.Db.PartCatalogueRequests.Add(r);
        await e.Db.SaveChangesAsync();
        return r;
    }

    private static async Task<MaterialItem> AddMaterialAsync(Env e, string code, string? mpn = null, string? mfr = null, string desc = "item", bool active = true)
    {
        var m = new MaterialItem { MaterialCode = code, Description = desc, ManufacturerPartNumber = mpn, Manufacturer = mfr, IsActive = active };
        e.Db.MaterialItems.Add(m);
        await e.Db.SaveChangesAsync();
        return m;
    }

    // ───────────────────────── Rules ─────────────────────────

    [Theory]
    [InlineData("ab-123 / x", "AB123X")]

    public void NormalizePartNumber_StripsSeparators_AndUppercases(string? input, string expected)
        => Assert.Equal(expected, PartCatalogueRules.NormalizePartNumber(input));

        => Assert.Equal(expected, PartCatalogueRules.IsMatchablePartNumber(value));

    [Fact]
    public void CodesMatch_IgnoresCaseAndEdgeWhitespace_ButNotStructure()
    {
        Assert.True(PartCatalogueRules.CodesMatch(" pn-1000 ", "PN-1000"));
        Assert.False(PartCatalogueRules.CodesMatch("PN1000", "PN-1000"));
        Assert.False(PartCatalogueRules.CodesMatch(null, "PN-1000"));
    }

    // ───────────────────────── Duplicate candidates ─────────────────────────

    [Fact]
    public async Task Duplicates_FindSameMpnDespiteFormatting_AndFlagManufacturer()
    {
        var e = NewEnv();
        await AddRequestAsync(e, "Completed", partNo: "PN-1", mpn: "6ES7 214-1AG40", mfrName: "Siemens");

        var same = await e.Dup.FindDuplicateCandidatesAsync(new PcDuplicateProbe { ManufacturerPartNumber = "6es7214-1ag40", NewManufacturerName = "SIEMENS" });
        var other = await e.Dup.FindDuplicateCandidatesAsync(new PcDuplicateProbe { ManufacturerPartNumber = "6ES7214-1AG40", NewManufacturerName = "ABB" });
        var unknown = await e.Dup.FindDuplicateCandidatesAsync(new PcDuplicateProbe { ManufacturerPartNumber = "6ES7214-1AG40" });

        Assert.Equal("SameMpnSameManufacturer", Assert.Single(same.Candidates).MatchKind);
        Assert.Equal("SameMpnOtherManufacturer", Assert.Single(other.Candidates).MatchKind);
        Assert.Equal("SameMpnManufacturerUnknown", Assert.Single(unknown.Candidates).MatchKind);
        Assert.False(same.IsAmbiguous);
        Assert.False(same.AutoMergeAllowed);
    }

    [Fact]
    public async Task Duplicates_NeverMatchPlaceholderMpns()
    {
        var e = NewEnv();
        await AddRequestAsync(e, "Completed", partNo: "PN-1", mpn: "NO_PART_NR");
        await AddRequestAsync(e, "Completed", partNo: "PN-2", mpn: "NO_PART_NR");

        var r = await e.Dup.FindDuplicateCandidatesAsync(new PcDuplicateProbe { ManufacturerPartNumber = "NO_PART_NR" });

        Assert.Empty(r.Candidates);
    }

    [Fact]
    public async Task Duplicates_TwoDistinctStrongParts_AreAmbiguous_AndNothingIsMerged()
    {
        var e = NewEnv();
        await AddRequestAsync(e, "Completed", partNo: "PN-A", mpn: "XYZ-100", mfrName: "ACME");
        await AddRequestAsync(e, "Completed", partNo: "PN-B", mpn: "XYZ100", mfrName: "Acme");

        var r = await e.Dup.FindDuplicateCandidatesAsync(new PcDuplicateProbe { ManufacturerPartNumber = "XYZ 100", NewManufacturerName = "ACME" });

        Assert.Equal(2, r.Candidates.Count);
        Assert.True(r.IsAmbiguous);
        Assert.False(r.AutoMergeAllowed);
        Assert.Equal(2, await e.Db.PartCatalogueRequests.CountAsync());   // هیچ رکوردی تغییر/حذف نشد
    }

    [Fact]
    public async Task Duplicates_MatchOldMpn_MaterialMaster_AndDoNotDoubleCountALinkedPart()
    {
        var e = NewEnv();
        var mi = await AddMaterialAsync(e, "PN-LINKED", mpn: "LINK-777", mfr: "ACME");
        var linked = await AddRequestAsync(e, "Completed", partNo: "PN-LINKED", mpn: "LINK-777", mfrName: "ACME");
        linked.MaterialItemId = mi.MaterialItemId;
        await AddRequestAsync(e, "Completed", partNo: "PN-OLD", mpn: "NEWCODE-1", oldMpn: "OLDCODE-9");
        await AddMaterialAsync(e, "MM-ONLY", mpn: "MMONLY-5", mfr: "ACME");
        await e.Db.SaveChangesAsync();

        var linkedHit = await e.Dup.FindDuplicateCandidatesAsync(new PcDuplicateProbe { ManufacturerPartNumber = "LINK777", NewManufacturerName = "ACME" });
        var oldHit = await e.Dup.FindDuplicateCandidatesAsync(new PcDuplicateProbe { ManufacturerPartNumber = "OLDCODE9" });
        var mmHit = await e.Dup.FindDuplicateCandidatesAsync(new PcDuplicateProbe { ManufacturerPartNumber = "MMONLY5", NewManufacturerName = "Acme" });

        Assert.Single(linkedHit.Candidates);                 // درخواست + MaterialItem پیوندخورده = یک قطعه
        Assert.False(linkedHit.IsAmbiguous);
        Assert.Equal("OldManufacturerPartNumber", Assert.Single(oldHit.Candidates).MatchedOn);
        var mm = Assert.Single(mmHit.Candidates);
        Assert.Equal("MaterialItem", mm.Source);
        Assert.Equal("SameMpnSameManufacturer", mm.MatchKind);
    }

    [Fact]
    public async Task Duplicates_DoNotDisclosePendingRequests_OnlyCountThem_AndExcludeSelf()
    {
        var e = NewEnv();
        var mine = await AddRequestAsync(e, "Draft", mpn: "PEND-100");
        await AddRequestAsync(e, "InApproval", mpn: "PEND-100", requester: OtherEmployee);
        await AddRequestAsync(e, "Cancelled", mpn: "PEND-100", requester: OtherEmployee);

        var r = await e.Dup.FindDuplicateCandidatesAsync(new PcDuplicateProbe { ManufacturerPartNumber = "PEND100", ExcludeRequestId = mine.PartCatalogueRequestId });

        Assert.Empty(r.Candidates);
        Assert.Equal(1, r.InFlightRequestsWithSameMpn);
    }

    [Fact]
    public async Task Search_RequiresAtLeastTwoCharacters_AndOnlyReturnsApprovedParts()
    {
        var e = NewEnv();
        await AddRequestAsync(e, "Completed", partNo: "VALVE-001", mpn: "V-100");
        await AddRequestAsync(e, "Draft", partNo: null, mpn: "V-100");
        await AddMaterialAsync(e, "MM-VALVE-9", desc: "Gate valve 2in");

        Assert.Empty(await e.Dup.SearchExistingPartsAsync("v"));
        Assert.Empty(await e.Dup.SearchExistingPartsAsync("   "));

        var byCode = await e.Dup.SearchExistingPartsAsync("valve 001");
        Assert.Equal("VALVE-001", Assert.Single(byCode).PartNumber);
        Assert.Equal("Exact", byCode[0].MatchKind);

        var byDesc = await e.Dup.SearchExistingPartsAsync("gate valve");
        Assert.Equal("MM-VALVE-9", Assert.Single(byDesc).PartNumber);
    }

    [Fact]
    public async Task Search_MatchesFreeFormatText_AndPrefersExactOverPartial()
    {
        var e = NewEnv();
        await AddRequestAsync(e, "Completed", partNo: "ZZ-1", description: "BALL VALVE CARBON STEEL");
        await AddRequestAsync(e, "Completed", partNo: "AB-12", mpn: "AB12");
        await AddRequestAsync(e, "Completed", partNo: "AB-123", mpn: "AB123X");

        var text = await e.Dup.SearchExistingPartsAsync("carbon steel");
        Assert.Equal("ZZ-1", Assert.Single(text).PartNumber);

        var ids = await e.Dup.SearchExistingPartsAsync("ab12");

        Assert.Equal("Exact", ids[0].MatchKind);
        Assert.Contains(ids, h => h.PartNumber == "AB-123" && h.MatchKind == "Partial");
    }

    // ───────────────────────── Material link ─────────────────────────

    [Fact]
    public async Task Link_Succeeds_OnExactCode_IsIdempotent_AndAppendsHistory()
    {
        var e = NewEnv();
        var mi = await AddMaterialAsync(e, "PN-1000");
        var req = await AddRequestAsync(e, "Completed", partNo: "pn-1000");

        Assert.Equal(mi.MaterialItemId, (await e.Trace.FindLinkCandidateAsync(req.PartCatalogueRequestId, e.Cataloguer))!.Id);

        var first = await e.Trace.LinkMaterialItemAsync(req.PartCatalogueRequestId, mi.MaterialItemId, e.Cataloguer);
        var again = await e.Trace.LinkMaterialItemAsync(req.PartCatalogueRequestId, mi.MaterialItemId, e.Cataloguer);

        Assert.True(first.Success);
        Assert.True(again.Success);
        Assert.Equal(mi.MaterialItemId, (await e.Db.PartCatalogueRequests.AsNoTracking().SingleAsync()).MaterialItemId);
        var history = await e.Db.PartCatalogueRequestApprovals.Where(a => a.Action == PartCatalogueActions.LinkMaterialItem).ToListAsync();
        Assert.Single(history);                                         // یک ردیف تاریخچه (Idempotent)
        Assert.Equal(e.Cataloguer.EmployeeId, history[0].ActorEmployeeId);
    }

    [Fact]
    public async Task Link_Rejects_CodeMismatch_NotCompleted_Inactive_Unauthorized_AndReplacement()
    {
        var e = NewEnv();
        var good = await AddMaterialAsync(e, "PN-2000");
        var other = await AddMaterialAsync(e, "PN-2001");
        var inactive = await AddMaterialAsync(e, "PN-2002", active: false);
        var done = await AddRequestAsync(e, "Completed", partNo: "PN-2000");
        var draft = await AddRequestAsync(e, "Draft", partNo: "PN-2000");
        var inactiveReq = await AddRequestAsync(e, "Completed", partNo: "PN-2002");

        Assert.Contains((await e.Trace.LinkMaterialItemAsync(done.PartCatalogueRequestId, other.MaterialItemId, e.Cataloguer)).Errors, x => x.Key == "Pc_Err_LinkCodeMismatch");
        Assert.Contains((await e.Trace.LinkMaterialItemAsync(draft.PartCatalogueRequestId, good.MaterialItemId, e.Cataloguer)).Errors, x => x.Key == "Pc_Err_LinkNotCompleted");
        Assert.Contains((await e.Trace.LinkMaterialItemAsync(inactiveReq.PartCatalogueRequestId, inactive.MaterialItemId, e.Cataloguer)).Errors, x => x.Key == "Pc_Err_InvalidReference");
        Assert.Contains((await e.Trace.LinkMaterialItemAsync(done.PartCatalogueRequestId, 99999, e.Cataloguer)).Errors, x => x.Key == "Pc_Err_InvalidReference");
        Assert.Contains((await e.Trace.LinkMaterialItemAsync(done.PartCatalogueRequestId, good.MaterialItemId, e.Nobody)).Errors, x => x.Key == "Pc_Err_NotAuthorized");
        Assert.Null(await e.Trace.FindLinkCandidateAsync(done.PartCatalogueRequestId, e.Nobody));
        Assert.Null((await e.Db.PartCatalogueRequests.AsNoTracking().FirstAsync(r => r.PartCatalogueRequestId == done.PartCatalogueRequestId)).MaterialItemId);

        // پیوند معتبر، سپس تلاش برای جایگزینی با MaterialItem دیگر مسدود است
        Assert.True((await e.Trace.LinkMaterialItemAsync(done.PartCatalogueRequestId, good.MaterialItemId, e.Cataloguer)).Success);
        var replace = await e.Trace.LinkMaterialItemAsync(done.PartCatalogueRequestId, other.MaterialItemId, e.Cataloguer);
        Assert.False(replace.Success);
        Assert.Equal(good.MaterialItemId, (await e.Db.PartCatalogueRequests.AsNoTracking().FirstAsync(r => r.PartCatalogueRequestId == done.PartCatalogueRequestId)).MaterialItemId);
    }

    // ───────────────────────── Demand origin ─────────────────────────

    private static async Task<int> AddWorkOrderAsync(Env e, string status = "Open")
    {
        var eq = new Equipment { TagNumber = "EQ-" + Guid.NewGuid().ToString("N")[..6], Name = "Pump", IsActive = true };
        e.Db.Equipments.Add(eq);
        await e.Db.SaveChangesAsync();
        var wo = new WorkOrder { EquipmentId = eq.EquipmentId, Status = status };
        e.Db.WorkOrders.Add(wo);
        await e.Db.SaveChangesAsync();
        return wo.WorkOrderId;
    }

    [Fact]
    public async Task DemandOrigin_AcceptsEachTypedOrigin_AndKeepsAtMostOne()
    {
        var e = NewEnv();
        var woId = await AddWorkOrderAsync(e);
        var sr = new StoreRequisition { Status = StoreRequisitionStatuses.Approved, IsActive = true };
        var mr = new MaterialRequisition { Status = "Submitted", IsActive = true };
        e.Db.StoreRequisitions.Add(sr);
        e.Db.MaterialRequisitions.Add(mr);
        await e.Db.SaveChangesAsync();
        var id = (await AddRequestAsync(e, "Draft")).PartCatalogueRequestId;

        Assert.True((await e.Trace.SetDemandOriginAsync(id, new PcDemandOriginInput { Type = "WorkOrder", Id = woId }, e.Requester())).Success);
        Assert.True((await e.Trace.SetDemandOriginAsync(id, new PcDemandOriginInput { Type = "storerequisition", Id = sr.StoreRequisitionId }, e.Requester())).Success);
        var afterSr = await e.Db.PartCatalogueRequests.AsNoTracking().SingleAsync();
        Assert.Null(afterSr.SourceWorkOrderId);                          // جایگزین شد، نه انباشته
        Assert.Equal(sr.StoreRequisitionId, afterSr.SourceStoreRequisitionId);

        Assert.True((await e.Trace.SetDemandOriginAsync(id, new PcDemandOriginInput { Type = "MaterialRequisition", Id = mr.MRId }, e.Requester())).Success);
        var afterMr = await e.Db.PartCatalogueRequests.AsNoTracking().SingleAsync();
        Assert.Null(afterMr.SourceStoreRequisitionId);
        Assert.Equal(mr.MRId, afterMr.SourceMaterialRequisitionId);

        Assert.True((await e.Trace.SetDemandOriginAsync(id, null, e.Requester())).Success);   // پاک‌کردن
        Assert.Null((await e.Db.PartCatalogueRequests.AsNoTracking().SingleAsync()).SourceMaterialRequisitionId);
    }

    [Fact]
    public async Task DemandOrigin_RejectsUnknownType_MissingOrClosedOrigin()
    {
        var e = NewEnv();
        var closedWo = await AddWorkOrderAsync(e, "Closed");
        var cancelledSr = new StoreRequisition { Status = StoreRequisitionStatuses.Cancelled, IsActive = true };
        var inactiveMr = new MaterialRequisition { Status = "Draft", IsActive = false };
        e.Db.StoreRequisitions.Add(cancelledSr);
        e.Db.MaterialRequisitions.Add(inactiveMr);
        await e.Db.SaveChangesAsync();
        var id = (await AddRequestAsync(e, "Draft")).PartCatalogueRequestId;

        Assert.Contains((await e.Trace.SetDemandOriginAsync(id, new PcDemandOriginInput { Type = "Equipment", Id = 1 }, e.Requester())).Errors, x => x.Key == "Pc_Err_OriginTypeInvalid");
        Assert.Contains((await e.Trace.SetDemandOriginAsync(id, new PcDemandOriginInput { Type = "WorkOrder", Id = 424242 }, e.Requester())).Errors, x => x.Key == "Pc_Err_InvalidReference");
        Assert.Contains((await e.Trace.SetDemandOriginAsync(id, new PcDemandOriginInput { Type = "WorkOrder", Id = 0 }, e.Requester())).Errors, x => x.Key == "Pc_Err_InvalidReference");
        Assert.Contains((await e.Trace.SetDemandOriginAsync(id, new PcDemandOriginInput { Type = "WorkOrder", Id = closedWo }, e.Requester())).Errors, x => x.Key == "Pc_Err_OriginClosed");
        Assert.Contains((await e.Trace.SetDemandOriginAsync(id, new PcDemandOriginInput { Type = "StoreRequisition", Id = cancelledSr.StoreRequisitionId }, e.Requester())).Errors, x => x.Key == "Pc_Err_OriginClosed");
        Assert.Contains((await e.Trace.SetDemandOriginAsync(id, new PcDemandOriginInput { Type = "MaterialRequisition", Id = inactiveMr.MRId }, e.Requester())).Errors, x => x.Key == "Pc_Err_InvalidReference");

        var r = await e.Db.PartCatalogueRequests.AsNoTracking().SingleAsync();
        Assert.Null(r.SourceWorkOrderId);
        Assert.Null(r.SourceStoreRequisitionId);
        Assert.Null(r.SourceMaterialRequisitionId);
    }

    [Fact]
    public async Task DemandOrigin_RequiresViewOnOriginModule_OwnerAndEditableStatus()
    {
        var e = NewEnv();
        var woId = await AddWorkOrderAsync(e);
        var draft = await AddRequestAsync(e, "Draft");
        var submitted = await AddRequestAsync(e, "InApproval");
        var input = new PcDemandOriginInput { Type = "WorkOrder", Id = woId };

        Assert.Contains((await e.Trace.SetDemandOriginAsync(draft.PartCatalogueRequestId, input, e.Requester("REQNOVIEW"))).Errors, x => x.Key == "Pc_Err_OriginNotAuthorized");
        Assert.Contains((await e.Trace.SetDemandOriginAsync(draft.PartCatalogueRequestId, input, new PcActor(OtherEmployee, 5, new[] { "REQ" }))).Errors, x => x.Key == "Pc_Err_NotEditable");
        Assert.Contains((await e.Trace.SetDemandOriginAsync(submitted.PartCatalogueRequestId, input, e.Requester())).Errors, x => x.Key == "Pc_Err_NotEditable");
        Assert.Contains((await e.Trace.SetDemandOriginAsync(draft.PartCatalogueRequestId, input, e.Nobody)).Errors, x => x.Key == "Pc_Err_NotEditable");

        Assert.All(await e.Db.PartCatalogueRequests.AsNoTracking().ToListAsync(), r => Assert.Null(r.SourceWorkOrderId));
    }
